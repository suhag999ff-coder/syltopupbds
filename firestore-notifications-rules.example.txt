// Notification-specific Firestore rules example.
// Merge this match block into your EXISTING rules; do not replace your site's rules wholesale.
// Admin Firebase Auth users need the custom claim { admin: true }.
match /notifications/{notificationId} {
  allow create, update, delete: if request.auth != null
    && request.auth.token.admin == true;

  allow read: if request.auth != null
    && (request.auth.token.admin == true
        || resource.data.audience == 'all'
        || resource.data.targetUid == request.auth.uid);
}
